async function fetchRealStockData(stockId) { addLog("▶ 正在抓取真實 K 線資料：" + stockId); const months = getRecentMonths(6); let allData = []; for (let i = 0; i < months.length; i++) { const yyyymmdd = months[i] + "01"; let monthData = await fetchTwseStockMonth(stockId, yyyymmdd); if (monthData.length === 0) { monthData = await fetchTpexStockMonth(stockId, yyyymmdd); } allData = allData.concat(monthData); await sleep(450); } allData.sort(function(a, b) { return new Date(a.time) - new Date(b.time); }); const unique = []; const map = {}; for (let i = 0; i < allData.length; i++) { const d = allData[i]; if (!map[d.time]) { map[d.time] = true; unique.push(d); } } if (unique.length < 20) { throw new Error("抓不到足夠真實K線資料，請確認股票代號是否正確，或該股票是否為上市/上櫃。"); } addLog("✓ 真實 K 線資料抓取完成，共 " + unique.length + " 筆"); return unique.slice(-120); } async function fetchTwseStockMonth(stockId, yyyymmdd) { try { const url = "https://www.twse.com.tw/exchangeReport/STOCK_DAY?response=json&date=" + yyyymmdd + "&stockNo=" + stockId; const response = await fetch(url); const json = await response.json(); if (!json || !json.data || !Array.isArray(json.data)) { return []; } const result = []; for (let i = 0; i < json.data.length; i++) { const row = json.data[i]; const date = rocDateToAD(row[0]); const open = parsePrice(row[3]); const high = parsePrice(row[4]); const low = parsePrice(row[5]); const close = parsePrice(row[6]); if ( date && isFinite(open) && isFinite(high) && isFinite(low) && isFinite(close) ) { result.push({ time: date, open: open, high: high, low: low, close: close }); } } if (result.length > 0) { addLog("✓ TWSE 上市資料：" + yyyymmdd + "，" + result.length + " 筆"); } return result; } catch (err) { addLog("⚠ TWSE 抓取失敗：" + err.message, true); return []; } } async function fetchTpexStockMonth(stockId, yyyymmdd) { try { const rocDate = adDateToRocDate(yyyymmdd); const url = "https://www.tpex.org.tw/www/zh-tw/afterTrading/tradingStock?code=" + stockId + "&date=" + rocDate + "&id=&response=json"; const response = await fetch(url); const json = await response.json(); let rows = []; if (json && Array.isArray(json.tables) && json.tables[0] && Array.isArray(json.tables[0].data)) { rows = json.tables[0].data; } else if (json && Array.isArray(json.data)) { rows = json.data; } else { return []; } const result = []; for (let i = 0; i < rows.length; i++) { const row = rows[i]; const date = rocDateToAD(row[0]); const open = parsePrice(row[3]); const high = parsePrice(row[4]); const low = parsePrice(row[5]); const close = parsePrice(row[6]); if ( date && isFinite(open) && isFinite(high) && isFinite(low) && isFinite(close) ) { result.push({ time: date, open: open, high: high, low: low, close: close }); } } if (result.length > 0) { addLog("✓ TPEx 上櫃資料：" + yyyymmdd + "，" + result.length + " 筆"); } return result; } catch (err) { addLog("⚠ TPEx 抓取失敗：" + err.message, true); return []; } } function getRecentMonths(count) { const arr = []; const now = new Date(); for (let i = count - 1; i >= 0; i--) { const d = new Date(now.getFullYear(), now.getMonth() - i, 1); const y = d.getFullYear(); const m = String(d.getMonth() + 1).padStart(2, "0"); arr.push(String(y) + m); } return arr; } function rocDateToAD(str) { if (!str) return ""; const clean = String(str).trim().replace(/\s/g, ""); const parts = clean.split("/"); if (parts.length !== 3) return ""; const year = parseInt(parts[0], 10) + 1911; const month = String(parseInt(parts[1], 10)).padStart(2, "0"); const day = String(parseInt(parts[2], 10)).padStart(2, "0"); return year + "-" + month + "-" + day; } function adDateToRocDate(yyyymmdd) { const year = parseInt(yyyymmdd.substring(0, 4), 10) - 1911; const month = yyyymmdd.substring(4, 6); const day = yyyymmdd.substring(6, 8); return year + "/" + month + "/" + day; } function parsePrice(value) { if (value === null || value === undefined) return NaN; const s = String(value) .replace(/,/g, "") .replace(/X/g, "") .replace(/x/g, "") .replace(/--/g, "") .trim(); if (s === "" || s === "---" || s === "--") return NaN; return Number(s); } function sleep(ms) { return new Promise(function(resolve) { setTimeout(resolve, ms); }); }
